GRAWP001
REALNAME -LabChat
IMAGE
SCRIPT
// Lab chat for 5.x/6.x, which have no chat bar of their own: type a
// line and press Enter to say it; Backspace deletes, Escape clears.
// keypressed gives #p(0) the virtual-key code and #p(1) the character.
if (created || playerenters) {
  setstring this.line,;
  showtext 200,8,screenheight-24,Arial,b,> ;
}
if (keypressed) {
  if (strequals(#p(0),13)) {
    setplayerprop #c,#s(this.line);
    setstring this.line,;
  } else if (strequals(#p(0),8)) {
    setstring this.line,#e(0,strlen(#s(this.line))-1,#s(this.line));
  } else if (strequals(#p(0),27)) {
    setstring this.line,;
  } else if (strlen(#p(1)) == 1) {
    setstring this.line,#s(this.line)#p(1);
  }
  showtext 200,8,screenheight-24,Arial,b,> #s(this.line);
}
SCRIPTEND
