# seihai-gserver settings: key = value, `#` starts a comment.
# Every line is optional; these are the defaults unless noted.

name = Seihai Lab
# Listen on loopback only. Set 0.0.0.0 to let other machines in.
bind = 127.0.0.1
port = 14802

# Where a new account starts, in tiles.
start_level = onlinestartlocal.nw
start_x = 30
start_y = 30.5

# Sent at login. An empty start_message sends none.
status_list = Online,Away,DND
start_message =

# Make an account on its first login with the password it gave.
# Handy in a lab; leave it off anywhere else.
create_accounts = false

# Seconds between world-time messages (keeps idle links busy; 0 = off).
heartbeat = 5
# Close a connection silent for this many seconds (0 = never).
idle_timeout = 0
login_timeout = 10
max_players = 256

# Stage 1 has no level files: the start board is one tile everywhere.
board_tile = 511

# The Seihai secure login (version token SEI00001): Seihai clients started
# with --secure-login send the account and password encrypted. This file
# holds the server's own key, relative to the world folder; it is made on
# first start and its fingerprint is printed so players can check it. Keep
# it private and back it up. Empty: secure logins are refused.
secure_login_key = config/seihai-login.key
# Refuse logins that send the password in the clear. Real Graal clients
# cannot log in when this is on.
require_secure_login = false

# The NPC-server link (see PROTOCOL.md in seihai-npcserver).
#   off       no NPC-server (the default)
#   auto      start seihai-npcserver from next to this binary, with a fresh
#             token each time; nothing else to set up
#   external  wait for an NPC-server started by hand with npcserver_token
npcserver = off
npcserver_bind = 127.0.0.1
npcserver_port = 14810
