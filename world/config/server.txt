# The Seihai secure login (version token SEI00001): Seihai clients started
# with --secure-login send the account and password encrypted. This file
# holds the server's own key, relative to the world folder; it is made on
# first start and its fingerprint is printed so players can check it. Keep
# it private and back it up. Empty: secure logins are refused.
secure_login_key = config/seihai-login.key
# Refuse logins that send the password in the clear. Real Graal clients
# cannot log in when this is on.
require_secure_login = false

# The NPC-server link (see PROTOCOL.md in seihai-npcserver).
#   off       no NPC-server (the default)
#   auto      start seihai-npcserver from next to this binary, with a fresh
#             token each time; nothing else to set up
#   external  wait for an NPC-server started by hand with npcserver_token
